cmake_minimum_required(VERSION 3.24)
project(Provider31091Validation LANGUAGES CXX)
set(NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../source/native")
option(EXPECT_31091_SUPPORTED "Expect the 310.9.1 hotfix admission" ON)
add_executable(AdaProviderAudit ada.cpp "${NATIVE}/dlssg_provider_policy.cpp")
add_executable(ProviderIdentityAudit policy.cpp "${NATIVE}/dlssg_provider_policy.cpp")
add_executable(V12LoaderAudit loader.cpp)
foreach(target IN ITEMS AdaProviderAudit ProviderIdentityAudit V12LoaderAudit)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
        EXPECT_31091_SUPPORTED=$<BOOL:${EXPECT_31091_SUPPORTED}>)
    target_include_directories(${target} PRIVATE "${NATIVE}")
    target_link_libraries(${target} PRIVATE bcrypt version)
endforeach()
