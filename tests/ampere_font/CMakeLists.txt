cmake_minimum_required(VERSION 3.24)
project(AmpereFontValidation LANGUAGES CXX)
set(NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../source/native")
set(MFG_PROVIDER_FIXTURE "" CACHE FILEPATH "Exact real data-only provider fixture")
set(MFG_FONT_PROVIDER_LIST "" CACHE FILEPATH "Text file containing audited real provider paths")
if(NOT EXISTS "${MFG_PROVIDER_FIXTURE}" OR NOT EXISTS "${MFG_FONT_PROVIDER_LIST}")
    message(FATAL_ERROR "Explicit provider fixture and provider list are required")
endif()
set(COMMON "${NATIVE}/dlssg_provider_policy.cpp" "${NATIVE}/ampere_native_cache.cpp"
    "${NATIVE}/ampere_cuda_program.cpp" "${NATIVE}/ampere_mask_transform.cpp")
add_executable(AmpereFontAudit font.cpp ${COMMON})
add_executable(AmperePublicationAudit ../ampere_publication/publication.cpp ${COMMON})
add_executable(AmperePublicationReadback ../ampere_publication/readback.cpp ${COMMON})
add_executable(CudaProgramAudit ../provider_31091/cuda_program.cpp "${NATIVE}/ampere_cuda_program.cpp")
foreach(target IN ITEMS AmpereFontAudit AmperePublicationAudit AmperePublicationReadback CudaProgramAudit)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
        MFG_PUBLICATION_CANDIDATE=1 MFG_PUBLICATION_COMPLETE=1)
    target_include_directories(${target} PRIVATE "${NATIVE}")
    target_link_libraries(${target} PRIVATE bcrypt version)
endforeach()
enable_testing()
add_test(NAME cuda_complete_program_and_font COMMAND CudaProgramAudit)
add_test(NAME ampere_publication_transaction COMMAND AmperePublicationAudit "${MFG_PROVIDER_FIXTURE}")
add_test(NAME ampere_publication_readback COMMAND AmperePublicationReadback "${MFG_PROVIDER_FIXTURE}")
file(STRINGS "${MFG_FONT_PROVIDER_LIST}" providers)
set(index 0)
foreach(provider IN LISTS providers)
    if(NOT EXISTS "${provider}")
        message(FATAL_ERROR "Missing audited provider: ${provider}")
    endif()
    math(EXPR index "${index} + 1")
    add_test(NAME "font_provider_${index}" COMMAND AmpereFontAudit "${provider}")
endforeach()
